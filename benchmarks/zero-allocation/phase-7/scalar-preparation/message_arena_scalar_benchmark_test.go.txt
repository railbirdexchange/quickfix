package quickfix

import "testing"

// BenchmarkMessageArenaScalarMutation isolates existing-field scalar changes,
// including both external byte aliases and aliases that must stage before a
// resize moves their source. All construction and warmup precede timing.
func BenchmarkMessageArenaScalarMutation(benchmark *testing.B) {
	for _, scenario := range []string{"typed", "external_bytes", "overlapping_bytes"} {
		benchmark.Run(scenario, func(benchmark *testing.B) {
			message := newArenaTestMessage(benchmark, 512, 16)
			message.Body.SetString(11, "order-id")
			message.Body.SetString(55, "ABCDEFGHIJKLMNOP")
			message.Body.SetInt(38, 123456)
			external := []byte("ABCDEFGHIJKLMNOP")
			mutate := func(iteration int) {
				switch scenario {
				case "typed":
					message.Body.SetInt(38, 123456+iteration%2)
					message.Body.SetBool(43, iteration%2 == 0)
					if iteration%2 == 0 {
						message.Body.SetString(11, "order-id")
					} else {
						message.Body.SetString(11, "longer-order-id")
					}
				case "external_bytes":
					message.Body.SetBytes(11, external[:8+8*(iteration%2)])
				case "overlapping_bytes":
					value, reject := message.Body.GetBytes(55)
					if reject != nil {
						benchmark.Fatal(reject)
					}
					message.Body.SetBytes(11, value[:8+8*(iteration%2)])
				}
			}
			for iteration := 0; iteration < 128; iteration++ {
				mutate(iteration)
			}
			if err := message.constructionError(); err != nil {
				benchmark.Fatal(err)
			}
			benchmark.ReportAllocs()
			benchmark.ResetTimer()
			for iteration := 0; iteration < benchmark.N; iteration++ {
				mutate(iteration)
			}
			benchmark.StopTimer()
			if err := message.constructionError(); err != nil {
				benchmark.Fatal(err)
			}
			if _, reject := message.Body.GetBytes(11); reject != nil {
				benchmark.Fatal(reject)
			}
		})
	}
}
