func newReplayCapacityTestPair(t *testing.T, transport, application *datadictionary.DataDictionary, limits ParseLimits) (*MessageSlot, *MessageSlot) {
	t.Helper()
	full, err := NewMessageSlot(transport, application, limits)
	if err != nil {
		t.Fatal(err)
	}
	admission, err := NewMessageSlot(transport, application, limits)
	if err != nil {
		t.Fatal(err)
	}
	// Use the same restoration behavior as the session's shared replay slot.
	full.enableScopedCallbacks()
	admission.enableScopedCallbacks()
	return full, admission
}

func assertReplayCapacityParity(t *testing.T, full, admission *MessageSlot, wire []byte) error {
	t.Helper()
	want := full.Parse(wire)
	got := admission.checkReplayCapacity(wire)
	if reflect.TypeOf(got) != reflect.TypeOf(want) || got != nil && got.Error() != want.Error() {
		t.Fatalf("admission error %T(%v), full parse %T(%v), frame %q", got, got, want, want, wire)
	}
	if admission.Message() != nil {
		t.Fatal("capacity admission exposed an unindexed Message")
	}
	if admission.fieldsUsed != full.fieldsUsed {
		t.Fatalf("extracted-field high-water mark = %d, full parse %d", admission.fieldsUsed, full.fieldsUsed)
	}
	if len(wire) <= admission.limits.MaxFrameBytes {
		for _, fields := range []*FieldMap{&admission.message.Header.FieldMap, &admission.message.Body.FieldMap, &admission.message.Trailer.FieldMap} {
			if len(fields.fixed.tags) != 0 || len(fields.fixed.touched) != 0 {
				t.Fatal("capacity admission populated a section index")
			}
		}
	}
	if got == nil {
		if !reflect.DeepEqual(admission.message.fields, full.message.fields) || !bytes.Equal(admission.message.bodyBytes, full.message.bodyBytes) {
			t.Fatal("capacity admission changed extracted fields or body boundaries")
		}
	}
	return got
}

func assertReplayCapacityResetReleases(t *testing.T, slot *MessageSlot) {
	t.Helper()
	slot.Reset()
	if slot.Message() != nil || slot.fieldsUsed != 0 || len(slot.message.fields) != 0 || slot.message.rawMessage != nil || slot.message.bodyBytes != nil || slot.message.inbound != nil || slot.raw.Bytes() != nil {
		t.Fatal("reset retained an active frame or parse scope")
	}
	for _, value := range slot.parseFields[:cap(slot.parseFields)] {
		if value.tag != 0 || value.value != nil || value.bytes != nil {
			t.Fatal("reset retained an extracted field, including failed lookahead")
		}
	}
	for _, definition := range slot.parseGroupStack[:cap(slot.parseGroupStack)] {
		if definition != nil {
			t.Fatal("reset retained a group stack reference")
		}
	}
	for _, fields := range slot.scopedFields {
		if len(fields.fixed.tags) != 0 || len(fields.fixed.touched) != 0 {
			t.Fatal("reset retained active section indexes")
		}
		for _, entry := range fields.fixed.entries {
			if entry.state != fixedFieldVacant || entry.tag != 0 || entry.values != nil || entry.scalar[0].value != nil || entry.scalar[0].bytes != nil {
				t.Fatal("reset retained indexed field references")
			}
		}
	}
}

func TestReplayCapacityLimitsMatchFullParse(t *testing.T) {
	dictionary, err := datadictionary.Parse("spec/FIX44.xml")
	if err != nil {
		t.Fatal(err)
	}
	nested := buildFIXT11Wire("35=D\x01", "78=1\x0179=A\x01539=1\x01524=N\x01804=1\x01545=X\x01")
	header := buildFIXT11Wire("35=D\x01627=2\x01628=A\x01628=B\x01", "11=order\x01")
	xml := buildFIXT11Wire("35=D\x01212=5\x01213=a\x01b\x01c\x01", "")
	duplicate := buildFIXT11Wire("35=D\x01", "11=A\x0111=B\x01")
	for _, test := range []struct {
		name   string
		limits ParseLimits
		wire   []byte
		want   error
	}{
		{"nested_exact", ParseLimits{len(nested), 10, 3, 3}, nested, nil},
		{"frame", ParseLimits{len(nested) - 1, 10, 3, 3}, nested, ErrFrameCapacity},
		{"fields_and_group_boundary_lookahead", ParseLimits{len(nested), 9, 3, 3}, nested, ErrFieldCapacity},
		{"nested_entries", ParseLimits{len(nested), 10, 2, 3}, nested, ErrGroupCapacity},
		{"nested_depth", ParseLimits{len(nested), 10, 3, 2}, nested, ErrGroupDepth},
		{"declared_entries", ParseLimits{4096, 64, 3, 3}, buildFIXT11Wire("35=D\x01", "453=4\x01"), ErrGroupCapacity},
		{"observed_entries", ParseLimits{4096, 64, 1, 3}, buildFIXT11Wire("35=D\x01", "453=1\x01448=A\x01448=B\x01"), ErrGroupCapacity},
		{"overflowing_group_count", ParseLimits{4096, 64, 1, 3}, buildFIXT11Wire("35=D\x01", "453=18446744073709551617\x01"), ErrGroupCapacity},
		{"header_exact", ParseLimits{4096, 64, 2, 1}, header, nil},
		{"header_entries", ParseLimits{4096, 64, 1, 1}, header, ErrGroupCapacity},
		{"header_depth", ParseLimits{4096, 64, 2, 0}, header, ErrGroupDepth},
		{"xml_exact_actual_fields", ParseLimits{4096, 6, 0, 0}, xml, nil},
		{"xml_actual_fields", ParseLimits{4096, 5, 0, 0}, xml, ErrFieldCapacity},
		{"duplicate_exact", ParseLimits{4096, 6, 0, 0}, duplicate, nil},
		{"duplicate_still_counts", ParseLimits{4096, 5, 0, 0}, duplicate, ErrFieldCapacity},
		{"field_error_precedes_body_length", ParseLimits{4096, 4, 0, 0}, []byte("8=FIXT.1.1\x019=9999\x0135=D\x0111=A\x0110=000\x01"), ErrFieldCapacity},
	} {
		t.Run(test.name, func(t *testing.T) {
			full, admission := newReplayCapacityTestPair(t, dictionary, dictionary, test.limits)
			if got := assertReplayCapacityParity(t, full, admission, test.wire); got != test.want {
				t.Fatalf("error = %v, want %v", got, test.want)
			}
			assertReplayCapacityResetReleases(t, admission)
			assertReplayCapacityResetReleases(t, full)
		})
	}
}

func TestReplayCapacityDictionaryAndSectionParity(t *testing.T) {
	dictionary, err := datadictionary.Parse("spec/FIX44.xml")
	if err != nil {
		t.Fatal(err)
	}
	transport, err := datadictionary.Parse("spec/FIXT11.xml")
	if err != nil {
		t.Fatal(err)
	}
	application, err := datadictionary.Parse("spec/FIX50SP2.xml")
	if err != nil {
		t.Fatal(err)
	}
	trailerGroup := datadictionary.NewGroupFieldDef(datadictionary.NewFieldType("TrailerGroup", 9000, "NUMINGROUP"), false,
		[]datadictionary.MessagePart{datadictionary.NewFieldDef(datadictionary.NewFieldType("TrailerID", 9001, "STRING"), true)})
	custom := *dictionary
	custom.Trailer = datadictionary.NewMessageDef("Trailer", "", []datadictionary.MessagePart{trailerGroup})
	admin := buildFIXT11Wire("35=A\x0134=1\x0149=S\x0156=T\x0152=20231231-20:19:41\x01", "98=0\x01108=30\x011137=9\x01384=2\x01372=D\x01385=R\x01372=F\x01385=R\x01")
	for _, test := range []struct {
		name                   string
		transport, application *datadictionary.DataDictionary
		wire                   []byte
		entries, depth         int
		want                   error
	}{
		{"no_dictionary", nil, nil, buildFIXT11Wire("35=D\x01", "11=A\x01"), 0, 0, nil},
		{"fix44_application_only", nil, dictionary, []byte(allocationBaselineRepeatingGroupMessage), 8, 3, nil},
		{"nested_siblings", dictionary, dictionary, buildFIXT11Wire("35=D\x01", "78=2\x0179=A\x01539=1\x01524=N\x01804=1\x01545=X\x0180=10\x0179=B\x01539=0\x0180=20\x01453=1\x01448=P\x01447=D\x01452=1\x01528=I\x01"), 8, 3, nil},
		{"trailer_exact", &custom, dictionary, buildFIXT11Wire("35=D\x01", "11=order\x019000=1\x019001=T\x01"), 1, 1, nil},
		{"trailer_entries", &custom, dictionary, buildFIXT11Wire("35=D\x01", "11=order\x019000=2\x019001=T\x019001=U\x01"), 1, 1, ErrGroupCapacity},
		{"trailer_depth", &custom, dictionary, buildFIXT11Wire("35=D\x01", "11=order\x019000=1\x019001=T\x01"), 1, 0, ErrGroupDepth},
		{"fixt_admin_dual_dictionary", transport, application, admin, 2, 1, nil},
		{"fixt_admin_nil_application", transport, nil, admin, 2, 1, nil},
		{"fixt_admin_dual_dictionary_limit", transport, application, admin, 1, 1, ErrGroupCapacity},
		{"fixt_admin_nil_application_limit", transport, nil, admin, 1, 1, ErrGroupCapacity},
	} {
		t.Run(test.name, func(t *testing.T) {
			full, admission := newReplayCapacityTestPair(t, test.transport, test.application, ParseLimits{4096, 64, test.entries, test.depth})
			if got := assertReplayCapacityParity(t, full, admission, test.wire); got != test.want {
				t.Fatalf("error = %v, want %v", got, test.want)
			}
			assertReplayCapacityResetReleases(t, admission)
		})
	}
}

func TestReplayCapacityMalformedWireMatchesFullParse(t *testing.T) {
	dictionary, err := datadictionary.Parse("spec/FIX44.xml")
	if err != nil {
		t.Fatal(err)
	}
	for _, test := range []struct {
		name    string
		wire    []byte
		wantErr bool
	}{
		{"empty", nil, true},
		{"wrong_first_tag", []byte("9=5\x018=FIXT.1.1\x0135=D\x0110=000\x01"), true},
		{"missing_equals", buildFIXT11Wire("35=D\x01", "invalid\x01"), true},
		{"missing_tag", buildFIXT11Wire("35=D\x01", "=value\x01"), true},
		{"invalid_tag", buildFIXT11Wire("35=D\x01", "X=value\x01"), true},
		// The existing parser accepts this overflowing numeric tag. Preserve
		// that result here; capacity admission must not add a new rejection.
		{"tag_overflow_legacy_acceptance", buildFIXT11Wire("35=D\x01", "999999999999999999999999=value\x01"), false},
		{"invalid_body_length", []byte("8=FIXT.1.1\x019=X\x0135=D\x0110=000\x01"), true},
		{"excessive_body_length", []byte("8=FIXT.1.1\x019=999999\x0135=D\x0110=000\x01"), true},
		{"wrong_body_length", []byte("8=FIXT.1.1\x019=0\x0135=D\x0110=000\x01"), true},
		{"duplicate_body_length", buildFIXT11Wire("35=D\x019=1\x01", "11=A\x01"), true},
		{"trailing_fields", append(buildFIXT11Wire("35=D\x01", ""), []byte("11=after-checksum\x01")...), true},
		{"xml_length_overflow", buildFIXT11Wire("35=D\x01212=999999\x01213=x\x01", ""), true},
		{"xml_length_negative", buildFIXT11Wire("35=D\x01212=-1\x01213=x\x01", ""), true},
		{"xml_payload_truncated", []byte("8=FIXT.1.1\x019=20\x0135=D\x01212=4\x01213=ab"), true},
		// Count syntax remains the dictionary validator's decision. Admission
		// must not turn parser success into rejection while suppressing indexes.
		{"empty_group_count", buildFIXT11Wire("35=D\x01", "453=\x01"), false},
		{"negative_group_count", buildFIXT11Wire("35=D\x01", "453=-1\x01"), false},
		{"invalid_group_count", buildFIXT11Wire("35=D\x01", "453=999X\x01"), false},
	} {
		t.Run(test.name, func(t *testing.T) {
			full, admission := newReplayCapacityTestPair(t, dictionary, dictionary, ParseLimits{4096, 64, 8, 3})
			if got := assertReplayCapacityParity(t, full, admission, test.wire); (got != nil) != test.wantErr {
				t.Fatalf("error = %v, want error %t", got, test.wantErr)
			}
			assertReplayCapacityResetReleases(t, admission)
		})
	}
}

func TestReplayCapacityEveryTruncatedPrefixMatchesFullParse(t *testing.T) {
	dictionary, err := datadictionary.Parse("spec/FIX44.xml")
	if err != nil {
		t.Fatal(err)
	}
	for _, wire := range [][]byte{
		buildFIXT11Wire("35=D\x01", "78=1\x0179=A\x01539=1\x01524=N\x01804=1\x01545=X\x01"),
		buildFIXT11Wire("35=D\x01212=5\x01213=a\x01b\x01c\x01", "11=A\x01"),
	} {
		full, admission := newReplayCapacityTestPair(t, dictionary, dictionary, ParseLimits{4096, 64, 8, 3})
		for end := 0; end < len(wire); end++ {
			if err := assertReplayCapacityParity(t, full, admission, wire[:end]); err == nil {
				t.Fatalf("accepted truncated prefix %d of %q", end, wire)
			}
			assertReplayCapacityResetReleases(t, admission)
			assertReplayCapacityResetReleases(t, full)
		}
	}
}

func TestReplayCapacityReuseAndResetReleaseLookahead(t *testing.T) {
	group := datadictionary.NewGroupFieldDef(datadictionary.NewFieldType("Group", 9000, "NUMINGROUP"), false,
		[]datadictionary.MessagePart{
			datadictionary.NewFieldDef(datadictionary.NewFieldType("Entry", 9001, "STRING"), true),
			datadictionary.NewFieldDef(datadictionary.NewFieldType("XMLDataLen", 212, "LENGTH"), false),
		})
	dictionary := &datadictionary.DataDictionary{Messages: map[string]*datadictionary.MessageDef{
		"D": datadictionary.NewMessageDef("WithGroup", "D", []datadictionary.MessagePart{group}),
	}}
	full, admission := newReplayCapacityTestPair(t, nil, dictionary, ParseLimits{1024, 16, 1, 1})
	valid := buildFIXT11Wire("35=D\x01", "9000=1\x019001=entry\x0155=tail\x01")
	if err := assertReplayCapacityParity(t, full, admission, valid); err != nil {
		t.Fatal(err)
	}
	invalid := buildFIXT11Wire("35=D\x01", "9000=1\x019001=entry\x01212=invalid\x0155=lookahead\x01")
	if err := assertReplayCapacityParity(t, full, admission, invalid); err != ErrInvalidUnsignedValue {
		t.Fatalf("group lookahead error: %v", err)
	}
	if string(admission.parseFields[:cap(admission.parseFields)][6].value) != "lookahead" {
		t.Fatal("fixture did not retain the extracted, rewound boundary field")
	}
	used := admission.fieldsUsed
	short := buildFIXT11Wire("35=0\x01", "")
	if err := assertReplayCapacityParity(t, full, admission, short); err != nil {
		t.Fatal(err)
	}
	if admission.fieldsUsed != used || used <= len(admission.message.fields) {
		t.Fatal("shorter admission lost the failed parse's high-water mark")
	}
	// The same storage must immediately support real resend reconstruction.
	if err := admission.Parse(valid); err != nil {
		t.Fatalf("full parse after failed/shorter admission: %v", err)
	}
	message := admission.Message()
	if message == nil {
		t.Fatal("full parse did not restore Message visibility")
	}
	if value, reject := message.Body.GetBytes(55); reject != nil || string(value) != "tail" {
		t.Fatalf("full parse did not restore section lookup: %q, %v", value, reject)
	}
	if values, found := message.Body.lookup(9000); !found || len(values) != 2 || string(values[1].value) != "entry" {
		t.Fatal("full parse did not restore the repeating-group index")
	}
	if err := assertReplayCapacityParity(t, full, admission, short); err != nil {
		t.Fatalf("admission after full parse: %v", err)
	}
	assertReplayCapacityResetReleases(t, admission)
	assertReplayCapacityResetReleases(t, full)
	if err := admission.Parse(valid); err != nil || admission.Message() == nil {
		t.Fatalf("full parse after reset: %v", err)
	}
	assertReplayCapacityResetReleases(t, admission)
}
